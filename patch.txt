void CGltfModel::BuildAABB()
{
    m_modelData.aabb = GltfAABB();

    for (auto& submesh : m_modelData.submeshes)
    {
        submesh.aabb = GltfAABB();

        if (submesh.vertexCount == 0)
            continue;

        size_t endVertex = submesh.vertexOffset + submesh.vertexCount;
        for (size_t i = submesh.vertexOffset; i < endVertex && i < m_modelData.vertices.size(); ++i)
        {
            submesh.aabb.Expand(m_modelData.vertices[i].position);
            m_modelData.aabb.Expand(m_modelData.vertices[i].position);
        }
    }
